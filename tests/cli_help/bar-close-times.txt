usage: krx-quant bar-close-times [-h] [--day DAY | --tonight]

options:
  -h, --help  show this help message and exit
  --day DAY   the trading day, YYYY-MM-DD (default: the latest day with
              predictions, through today)
  --tonight   the day of the night it runs in (the date, or the one before
              until 06:00), as the scheduled job runs it; exit 1 when that day
              has no predictions
